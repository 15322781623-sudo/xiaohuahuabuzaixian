# 游戏登录页资源加载与自动更新修复方案

## 📋 问题诊断

### 当前项目 (game.html) 存在的问题
1. **资源加载路径错误**: 使用本地路径 `/game/` 而非 CDN 远程路径
2. **Bundle 版本锁定**: Cocos launcher bundle 会覆盖 CODE_VERSION 导致卡死
3. **CDN 资源加载器安装顺序不当**: convertAssets 包装时机不够早
4. **引擎缓存缺失**: 每次启动都需要下载 Cocos 引擎
5. **Manifest 获取失败**: 缺少远程 manifest 拉取机制
6. **深度预热缺失**: 场景加载阻塞主线程

### niaoge 项目的优秀设计
1. **CDN 优先策略**: 所有资源从 `https://xxz-xyzw.hortorgames.com/game/` 加载
2. **智能引擎缓存**: localStorage 命中→毫秒注入，未命中→同步 XHR 下载
3. **Manifest 版本锁定**: fetchRemoteBundleVers + localStorage 缓存
4. **深度预热机制**: preBoot() → 提前加载 bundle → 挂起在 cc.game.run()
5. **精细脚本队列**: 分主队列 + 延迟队列，确保引擎就绪后再执行

```javascript
<script>
(function() {
  var CDN = 'https://xxz-xyzw-res.hortorgames.com/remote/game/';
  var ENGINE_FILE = 'cocos2d-js-min.a5841.js';
  var CACHE_KEY = '__cocos_engine__' + ENGINE_FILE;

  // 1. 尝试从 localStorage 读取
  var cached = null;
  try { cached = localStorage.getItem(CACHE_KEY); } catch(e) {}

  if (cached && cached.length > 100000) {
    console.log('[SpeedBoost] ⚡ 引擎缓存命中');
    document.write('<script>' + cached + '<\/script>');
  } else {
    // 2. 缓存未命中：同步 XHR 下载 + 缓存
    console.log('[SpeedBoost] 缓存未命中，从 CDN 下载...');
    var xhr = new XMLHttpRequest();
    xhr.open('GET', CDN + ENGINE_FILE, false);
    xhr.send();
    if (xhr.status === 200) {
      var code = xhr.responseText;
      if (code.length > 100000) {
        try { localStorage.setItem(CACHE_KEY, code); } catch(e) {}
        console.log('[SpeedBoost] 引擎已缓存 (' + (code.length/1024).toFixed(0) + 'KB)');
        document.write('<script>' + code + '<\/script>');
      } else {
        document.write('<script src="' + CDN + ENGINE_FILE + '"><\/script>');
      }
    }
  }
})();
</script>
```

### 二、boot.js 完整重构

#### 2.1 新增 fetchRemoteBundleVers 函数
```javascript
// 从服务器获取最新 bundleVers（带 localStorage 缓存）
function fetchRemoteBundleVers(settings) {
  return new Promise(function(resolve, reject) {
    var CACHE_KEY = '__boot_manifest_cache__';
    var appliedCache = false;

    // ★ 优先应用本地缓存（秒开，避免阻塞）
    try {
      var cached = localStorage.getItem(CACHE_KEY);
      if (cached) {
        var cacheData = JSON.parse(cached);
        if (cacheData && cacheData.bundleVers) {
          Object.assign(settings.bundleVers, cacheData.bundleVers);
          appliedCache = true;
          console.log('[boot v1.0] ✅ 已应用 manifest 缓存');
        }
      }
    } catch(e) {}

    var xhr = new XMLHttpRequest();
    var manifestUrl = 'https://xxz-xyzw.hortorgames.com/login/manifest?platform=hortor&version=0.32.0-android';
    
    xhr.open('POST', manifestUrl, true);
    xhr.setRequestHeader('Content-Type', 'application/json;charset=UTF-8');
    xhr.timeout = 5000;

    xhr.onload = function() {
      if (xhr.status === 200) {
        try {
          var data = JSON.parse(xhr.responseText);
          var body = data.body;
          if (typeof body === 'string') body = JSON.parse(body);
          var bv = body && body.bundleVers;
          if (typeof bv === 'string') bv = JSON.parse(bv);

          if (bv) {
            // 缓存到 localStorage
            try {
              localStorage.setItem(CACHE_KEY, JSON.stringify({
                bundleVers: bv,
                time: Date.now()
              }));
            } catch(e) {}
            
            // 合并到 settings.bundleVers
            Object.assign(settings.bundleVers, bv);
            console.log('[boot v1.0] ✅ Manifest 获取成功!');
            
            // 自动同步服务器版本
            if (bv.codeVersion) {
              settings.codeVersion = bv.codeVersion;
            }
          }
        } catch(e) {
          console.warn('[boot v1.0] ⚠ Manifest 解析失败:', e.message);
        }
      }
      resolve(); // 总是 resolve（有缓存或本地版本兜底）
    };

    xhr.onerror = xhr.ontimeout = function() {
      if (!appliedCache) console.warn('[boot v1.0] ⚠ Manifest 网络错误');
      resolve();
    };

    xhr.send('{}');
  });
}
```

#### 2.2 新增 lockCodeVersion 函数（防止 launcher 覆盖版本号）
```javascript
function lockCodeVersion(settings) {
  var ver = settings.codeVersion || '2.29.2';
  var gameVer = globalThis.GAME_VERSION || '0.32.0-android';
  var commitId = globalThis.COMMIT_ID || '';

  try { delete globalThis.CODE_VERSION; } catch(e) {}
  try { delete window.CODE_VERSION; } catch(e) {}

  // 使用 defineProperty 防止被覆盖
  Object.defineProperty(globalThis, 'CODE_VERSION', {
    get: function() { return ver; },
    set: function(v) {
      console.warn('[boot v1.0] launcher 尝试覆盖 CODE_VERSION 为', v + '— 已阻止');
    },
    configurable: true
  });

  settings.codeVersion = ver;
  globalThis.GAME_VERSION = gameVer;
  globalThis.ENV = 'Prod';
  globalThis.GAME_ID = 'xyzw';
  
  console.log('[boot v1.0] ✅ 版本信息已锁定:', ver);
}
```

```javascript
// ★ preBoot() - 深度预热（只加载到 bundle 就绪，暂停在 cc.game.run() 前）
window.preBoot = async function() {
  var settings = window._CCSettings;
  await fetchRemoteBundleVers(settings);
  lockCodeVersion(settings);
  
  if (typeof window.installRemoteAssetLoader === 'function') {
    window.installRemoteAssetLoader();

  doStartGame(settings, true); // true = deepPreload mode

// ★ boot() - 完整启动（正常模式）
window.boot = async function() {
  var settings = window._CCSettings;
  await fetchRemoteBundleVers(settings);
  lockCodeVersion(settings);
  
  if (typeof window.installRemoteAssetLoader === 'function') {
    window.installRemoteAssetLoader();
  }

  // 完整启动
  doStartGame(settings, false); // false = normal mode
};
```

#### 5.1 game/smart-engine-loader.js
```javascript
// 智能引擎加载器 - 借鉴 niaoge 实现
(function() {
  var CDN = 'https://xxz-xyzw-res.hortorgames.com/remote/game/';
  var ENGINE_FILE = 'cocos2d-js-min.a5841.js';
  var CACHE_KEY = '__cocos_engine__' + ENGINE_FILE;

  function loadFromCache() {
    try {
      var cached = localStorage.getItem(CACHE_KEY);
      if (cached && cached.length > 100000) {
        console.log('[SmartEngineLoader] ⚡ 引擎缓存命中');
        document.write('<script>' + cached + '<\/script>');
        return true;
      }
    } catch(e) {}
    return false;
  }

  function loadFromNetwork(callback) {
    var xhr = new XMLHttpRequest();
    xhr.open('GET', CDN + ENGINE_FILE, false);
    xhr.send();
    
    if (xhr.status === 200) {
      var code = xhr.responseText;
      if (code.length > 100000) {
        try { localStorage.setItem(CACHE_KEY, code); } catch(e) {}
        console.log('[SmartEngineLoader] 引擎已缓存 (' + (code.length/1024).toFixed(0) + 'KB)');
        document.write('<script>' + code + '<\/script>');
        callback && callback();
        return;
      }
    }
    
    // 回退：直接用 src 加载
    document.write('<script src="' + CDN + ENGINE_FILE + '"><\/script>');
    callback && callback();
  }

  if (!loadFromCache()) {
    loadFromNetwork();
  }
})();
```

#### 5.2 game/resource-patch.js
```javascript
// 强制所有 bundle 从 CDN 加载（不受 convertAssets 影响）
(function() {
  var CDN_BASE = "https://xxz-xyzw-res.hortorgames.com/remote/";
  
  if (!window.cc) {
    Object.defineProperty(window, 'cc', {
      configurable: true,
      set: function(ccObj) {
        installBundlePatch(ccObj);
        Object.defineProperty(window, 'cc', {
          configurable: true,
          writable: true,
          value: ccObj
        });
      }
    });
  } else {
    setTimeout(function() { installBundlePatch(window.cc); }, 100);
  }
  
  function installBundlePatch(cc) {
    setTimeout(function() {
      if (!cc.assetManager) {
        return installBundlePatch(cc);
      }
      
      var am = cc.assetManager;
      var origLoadBundle = am.loadBundle.bind(am);
      
      // 拦截 loadBundle - 强制所有 bundles 从 CDN 加载（排除 internal）
      am.loadBundle = function(bundleName, callback, userObject) {
        if (typeof bundleName === 'string' && 
            !bundleName.startsWith('http') && 
            !['internal', 'main'].includes(bundleName)) {
          
          var cdnUrl = CDN_BASE + bundleName;
          console.log('[ResourcePatch] Converting bundle:', bundleName, '→', cdnUrl);
          
          var options = null;
          if (am.bundleVers && am.bundleVers[bundleName]) {
            options = {version: am.bundleVers[bundleName]};
          }
          
          origLoadBundle(cdnUrl, callback, userObject);
        } else {
          console.log('[ResourcePatch] Keeping bundle local:', bundleName);
          origLoadBundle(bundleName, callback, userObject);
        }
      };
      
      // 同时 patch downloadFile
      var origDownload = am.downloader.download.bind(am.downloader);
      am.downloader.download = function(id, url, ext, options, onComplete) {
        if (typeof url === 'string' && url.startsWith('assets/')) {
          url = CDN_BASE + url.slice(7);
          console.log('[ResourcePatch] Converting file:', id, url);
        }
        origDownload(id, url, ext, options, onComplete);
      };
      
      console.log('[ResourcePatch] ✅ Bundle patch installed!');
    }, 200);
  }
})();
```

## 💡 关键设计思想

### 1. "CDN 优先" 而非"本地优先"
- 传统思路：先用本地，没有再网络
- **新思路**：始终网络优先，本地仅作兜底
- 原因：保证用户永远使用最新资源，无需更新客户端

### 2. "深度预热" 而非"启动时加载"
- 传统思路：点击才开始加载资源
- **新思路**：提前加载 bundle → 挂起 → 点击进入瞬间显示
- 原因：将 IO 等待转移到后台，用户体验零等待

### 3. "智能缓存" 而非"盲目下载"
- 传统思路：每次都重新下载
- **新思路**：localStorage 缓存引擎 + manifest → 秒开
- 原因：平衡"实时性"与"性能"，缓存命中率决定体验

### 4. "版本锁定" 而非"动态变更"
- 传统思路：launcher 随意覆盖版本号
- **新思路**：defineProperty 锁定 CODE_VERSION → 拒绝修改
- 原因：防止 launcher bundle 覆盖导致卡死
